let i = 0;
let answers = [];
let role = '';

const $ = selector => document.querySelector(selector);


document.querySelectorAll('.role').forEach(button => {

button.addEventListener('click', () => {

role =
button.textContent.trim() === 'Поставщика'
? 'supplier'
: 'buyer';

$('#start').classList.add('hidden');
$('#quiz').classList.remove('hidden');

draw();

});

});


function draw() {

const item = data[i];

$('#counter').textContent =
`Вопрос ${i + 1} из ${data.length}`;

$('#bar').style.width =
`${(i + 1) / data.length * 100}%`;

$('#cat').textContent = item.cat;
$('#question').textContent = item.q;
$('#hint').textContent = item.hint;

$('#answers').innerHTML = '';

[
['yes', 'Да'],
['no', 'Нет'],
['unknown', 'Не уверен']
].forEach(([value, label]) => {

const button = document.createElement('button');

button.className = 'btn';
button.type = 'button';
button.textContent = label;

button.addEventListener('click', () => {

answers.push({
...item,
answer:value
});

i++;

if (i < data.length) {
draw();
} else {
show();
}

});

$('#answers').appendChild(button);

});

}


function show() {

$('#quiz').classList.add('hidden');
$('#finish').classList.remove('hidden');

const bad =
answers.filter(item => item.answer !== 'yes');

const good =
answers.filter(item => item.answer === 'yes');

const side =
role === 'supplier'
? 'поставщика'
: 'покупателя';


$('#resultTitle').textContent =
bad.length
? `Что стоит исправить в интересах ${side}`
: 'По ответам явных проблем не нашли';


$('#resultIntro').textContent =
bad.length
? 'Рекомендации ниже меняются в зависимости от выбранной стороны договора.'
: 'Все проверенные условия определены. Это не исключает других рисков конкретного договора.';


$('#problems').innerHTML = '';


bad.forEach(item => {

const text = item[role];

const element =
document.createElement('article');

element.className = 'result';


element.innerHTML = `

<div class="result-head">

<h2 style="margin:0">
${item.name}
</h2>

<span class="badge ${
item.level === 'Высокий риск'
? 'high'
: 'medium'
}">
${item.level}
</span>

</div>


<p class="muted">
${
item.answer === 'unknown'
? 'Вы ответили: «Не уверен».'
: 'Вы ответили: «Нет».'
}
</p>


<div class="answer-block">

<p class="answer-label">
<strong>Чем грозит:</strong>
</p>

<p>
${text.danger}
</p>

</div>


<div class="answer-block">

<p class="answer-label">
<strong>Как исправить:</strong>
</p>

<p>
${text.fix}
</p>

</div>


<div class="practice">

<div class="practice-title">
Судебная практика
</div>

<div>
${item.practice}
</div>

<div class="court">
${item.court}
</div>

</div>

`;


$('#problems').appendChild(element);

});


if (good.length) {

$('#good').classList.remove('hidden');

$('#goodText').textContent =
good
.map(item => item.name)
.join(' · ');

} else {

$('#good').classList.add('hidden');

}

}


function reset() {

i = 0;
answers = [];
role = '';

$('#quiz').classList.add('hidden');
$('#finish').classList.add('hidden');
$('#start').classList.remove('hidden');

}


$('#reset').addEventListener('click', reset);
$('#resetTop').addEventListener('click', reset);
